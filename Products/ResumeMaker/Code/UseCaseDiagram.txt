@startuml

left to right direction

actor "User" as User

package "Resume Maker" {
  usecase "Move down entity" as MoveDownEntity
  usecase "Move up entity" as MoveUpEntity
  usecase "Delete entity" as DeleteEntity
  usecase "Update entity" as UpdateEntity
  usecase "Show entities" as ShowEntities
  usecase "Create entity" as CreateEntity
  usecase "Manage entity" as ManageEntity
  usecase "Generate PDF resume" as GeneratePDFResume
  usecase "Deselect resume" as DeselectResume
  usecase "Select resume" as SelectResume
  usecase "Manage experience" as ManageExperience
  usecase "Manage education" as ManageEducation
  usecase "Manage skill" as ManageSkill
  usecase "Manage resume" as ManageResume
}

User -- ManageExperience
User -- ManageEducation
User -- ManageSkill
User -- ManageResume

ManageResume <.. GeneratePDFResume : <<Extend>>
ManageResume <.. DeselectResume : <<Extend>>
ManageResume <.. SelectResume : <<Extend>>

ManageResume --|> ManageEntity
ManageSkill --|> ManageEntity
ManageEducation --|> ManageEntity
ManageExperience --|> ManageEntity

ManageEntity <.. MoveDownEntity : <<Extend>>
ManageEntity <.. MoveUpEntity : <<Extend>>
ManageEntity <.. DeleteEntity : <<Extend>>
ManageEntity <.. UpdateEntity : <<Extend>>
ManageEntity <.. ShowEntities : <<Extend>>
ManageEntity <.. CreateEntity : <<Extend>>

@enduml
